\textbf{Mathematical Framework:}

% \footnotesize
\textbf{State:} $\mathbf{x} = [x_{l1}, y_{l1}, v_{xl1}, v_{yl1}, x_{l2}, y_{l2}, x_f, y_f, v_{xf}, v_{yf}]^T$

\textbf{Motion:} $\mathbf{x}_{k} = \mathbf{F} \mathbf{x}_{k-1} + \mathbf{w}_{k-1}$

\textbf{Measurement:} $z_k = \sqrt{(x_f - x_{li})^2 + (y_f - y_{li})^2} + v_k$
